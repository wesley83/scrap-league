export async function journeys({api,frame,doc,win,home,click,button,check,waitFor,delay,connect}) {
 const trace=[];
 const change=(selector,value)=>{
  const el=doc().querySelector(selector);check(el,'Missing form control '+selector);
  el.value=String(value);el.dispatchEvent(new (win().Event)('change',{bubbles:true}));
  trace.push({action:'form-change',selector,value});
 };
 const activate=id=>{const el=doc().getElementById(id);check(el&&!el.disabled,'Missing/disabled action '+id);el.click();trace.push({action:'button',id});};
 const selectRoad=index=>change('#object-list',api().workshop.track.pieces[index].id);
 const blank=async name=>{
  await home();await api().openWorkshop();await api().workshop.flush();
  activate('work-file');await waitFor(()=>doc().querySelector('dialog').open&&doc().querySelector('[data-file="blank"]'));
  doc().querySelector('[data-file="blank"]').click();await waitFor(()=>api().workshop.track.pieces.length===0);
  trace.push({action:'new-blank'});change('#track-name',name);
 };
 const numericFields=(values)=>{for(const [key,value] of Object.entries(values)){
  const el=doc().querySelector('[data-field="'+key+'"]');check(el,'Missing inspector '+key);el.value=String(value);
  trace.push({action:'inspector-field',key,value});
 }activate('apply-object');};
 return {
  'Critical input and enlarged text layout matrix':async()=>{
   const a=api(),settings=structuredClone(a.settings),rows=[];
   const enlarge=root=>{
    const elements=[root,...root.querySelectorAll('*')],original=elements.map(el=>el.style.fontSize),sizes=elements.map(el=>parseFloat(win().getComputedStyle(el).fontSize));
    elements.forEach((el,i)=>el.style.fontSize=sizes[i]*2+'px');
    return {elements,sizes,restore:()=>elements.forEach((el,i)=>el.style.fontSize=original[i])};
   };
   try{
    await home();await a.openWorkshop();await a.workshop.flush();
    for(const [width,height]of [[320,568],[390,844],[844,390],[768,1024],[1024,768],[1440,900]]){
     frame.style.width=width+'px';frame.style.height=height+'px';await delay(80);
     check(win().innerWidth===width&&win().innerHeight===height,'Viewport change did not apply');
     const root=doc().querySelector('#screen-workshop'),zoom=enlarge(root);await delay(80);
     try{
      const field=doc().querySelector('#track-name');field.focus();field.scrollIntoView({block:'center'});await delay(30);
      const overflow=root.scrollWidth-root.clientWidth;
      check(overflow<=1,'Workshop 200% text horizontal overflow '+overflow+'px at '+width+'x'+height);
      check(doc().activeElement===field,'Workshop name focus lost');
      const fr=field.getBoundingClientRect();check(fr.left>=0&&fr.right<=width+1&&fr.top>=0&&fr.bottom<=height+1,'Enlarged Workshop name cannot be brought into view');
      rows.push({viewport:[win().innerWidth,win().innerHeight],surface:'Workshop',computedFontScale:2,measuredFieldFont:win().getComputedStyle(field).fontSize,overflow,focusedFieldVisible:true});
      for(const [panel,selector,label]of [['tools','#piece-kind','Tools'],['inspector','#object-list','Inspector']]){
       if(width<=700)[...doc().querySelectorAll('.editor-sheets button')].find(button=>button.textContent===label).click();
       const pane=root.querySelector(panel==='tools'?'.palette':'.inspector'),control=root.querySelector(selector);
       control.focus();control.scrollIntoView({block:'center'});await delay(30);
       const overflow=pane.scrollWidth-pane.clientWidth,r=control.getBoundingClientRect();
       check(overflow<=1,'Enlarged '+panel+' panel overflows '+overflow+'px at '+width+'x'+height);
       check(r.left>=0&&r.right<=width+1&&r.top>=0&&r.bottom<=height+1,'Enlarged '+panel+' control cannot be brought into view');
       rows.push({viewport:[win().innerWidth,win().innerHeight],surface:'Workshop '+panel,computedFontScale:2,overflow,focusedFieldVisible:true});
      }
      if(width<=700)[...doc().querySelectorAll('.editor-sheets button')].find(button=>button.textContent==='Track').click();
     }finally{zoom.restore();}
     click('settings-button');await waitFor(()=>doc().querySelector('#modal-title').textContent==='SETTINGS');
     const modal=doc().querySelector('dialog'),enlarged=enlarge(modal);await delay(80);
     try{
      const field=modal.querySelector('input:not([type=checkbox]):not([type=range]):not([type=file])');
      check(field,'Driver-name field missing');field.focus();field.scrollIntoView({block:'center'});await delay(30);
      const overflow=modal.scrollWidth-modal.clientWidth,r=field.getBoundingClientRect(),m=modal.getBoundingClientRect();
      check(overflow<=1,'Settings 200% text horizontal overflow '+overflow+'px at '+width+'x'+height);
      check(r.left>=m.left&&r.right<=m.right+1&&r.top>=m.top&&r.bottom<=m.bottom+1,'Enlarged Settings field cannot be brought into view');
      rows.push({viewport:[win().innerWidth,win().innerHeight],surface:'Settings',computedFontScale:2,measuredFieldFont:win().getComputedStyle(field).fontSize,overflow,focusedFieldVisible:true});
     }finally{enlarged.restore();modal.close();}
    }
    return {rows,scope:'Actual iframe CSS viewport and every computed descendant font explicitly doubled, with focused-field visibility/scroll checks. This is text-size emulation, not OS zoom, physical phone or a full WCAG certification.'};
   }finally{Object.assign(a.settings,settings);a.applyPreferences();frame.style.width='1280px';frame.style.height='720px';}
  },
  'Complete injected gamepad pause settings resume':async()=>{
   await home();const a=api(),nav=win().navigator,prior=Object.getOwnPropertyDescriptor(nav,'getGamepads');
   const pad={axes:[0,0],buttons:Array.from({length:17},()=>({pressed:false,value:0}))};
   const press=async n=>{pad.buttons[n]={pressed:true,value:1};await delay(120);pad.buttons[n]={pressed:false,value:0};await delay(250);};
   const settings=structuredClone(a.settings);
   Object.defineProperty(nav,'getGamepads',{configurable:true,value:()=>[pad]});
   try{
    a.settings.autoAccelerate=false;a.settings.assist=false;a.settings.sensitivity=1;a.applyPreferences();
    click('quick-race');await waitFor(()=>a.race?.state==='racing');
    pad.axes[0]=0.7;pad.buttons[7]={pressed:true,value:0.8};await delay(180);
    const before=a.input();check(before.steer>0.5&&before.throttle>0.5,'Gamepad driving not accepted');
    pad.axes[0]=0;pad.buttons[7]={pressed:false,value:0};await press(9);check(a.race.state==='paused','Start did not pause');
    const clock=a.race.time;await press(13);check(doc().activeElement.textContent==='Settings','D-pad did not reach Settings');
    await press(0);check(doc().querySelector('#modal-title').textContent==='SETTINGS','A did not open nested settings');
    await press(1);check(doc().querySelector('#modal-title').textContent==='PAUSED','B did not return to pause');
    check(a.race.time===clock,'Race clock moved in nested settings');await press(0);await waitFor(()=>a.race.state==='racing');
    check(doc().activeElement.id==='world','Resume did not restore gameplay focus');
    pad.axes[0]=-0.7;pad.buttons[7]={pressed:true,value:0.8};await delay(180);
    const after=a.input();check(after.steer<-.5&&after.throttle>.5,'Fresh gamepad control lost after settings');
    pad.axes[0]=0;pad.buttons[7]={pressed:false,value:0};a.pause();
    return {before,after,clockStopped:true,menuSequence:['Start','D-pad down','A Settings','B Back to pause','A Resume'],scope:'Injected navigator gamepad through actual game loop and menu handlers; no physical controller/native desktop claim'};
   }finally{if(prior)Object.defineProperty(nav,'getGamepads',prior);else delete nav.getGamepads;Object.assign(a.settings,settings);a.applyPreferences();}
  },
  'Complete injected touch pause settings resume':async()=>{
   const a=api(),settings=structuredClone(a.settings),rows=[];
   try{
    for(const [width,height]of [[390,844],[844,390]])for(const layout of ['pad','buttons'])for(const leftHanded of [false,true]){
     await home();frame.style.width=width+'px';frame.style.height=height+'px';
     Object.assign(a.settings,{touch:true,touchLayout:layout,leftHanded,autoAccelerate:false,assist:false,sensitivity:1});a.applyPreferences();
     click('quick-race');await waitFor(()=>a.race?.state==='racing');
     const pad=doc().querySelector(layout==='pad'?'#steering-pad':'[data-hold="right"]'),gas=doc().querySelector('[data-hold="throttle"]');
     const restored=[],captures=[];
     for(const el of [pad,gas]){const method=el.setPointerCapture;el.setPointerCapture=id=>captures.push(id);restored.push(()=>el.setPointerCapture=method);}
     const event=(el,type,id,ratio=.85)=>{const r=el.getBoundingClientRect();el.dispatchEvent(new (win().PointerEvent)(type,{pointerId:id,pointerType:'touch',isPrimary:id===501,button:0,buttons:type==='pointerup'?0:1,clientX:r.left+r.width*ratio,clientY:r.top+r.height/2,bubbles:true,cancelable:true}));};
     try{
      event(pad,'pointerdown',501);event(gas,'pointerdown',502);const before=a.input();
      check(before.steer>.2&&before.throttle===1,'Two-contact touch driving not accepted');await delay(150);
      click('pause-button');check(a.race.state==='paused','Touch pause failed');const clock=a.race.time;
      const neutral=a.input();check(neutral.steer===0&&neutral.throttle===0,'Pause retained touch ownership');
      button('Settings').click();await waitFor(()=>doc().querySelector('#modal-title').textContent==='SETTINGS');
      button('Back to pause').click();check(a.race.time===clock,'Nested touch settings advanced race');button('Resume').click();await waitFor(()=>a.race.state==='racing');
      const resumed=a.input();check(resumed.steer===0&&resumed.throttle===0,'Resume replayed old touch contacts');
      event(pad,'pointerdown',503);event(gas,'pointerdown',504);const after=a.input();check(after.steer>.2&&after.throttle===1,'Fresh touch after settings was lost');
      event(pad,'pointerup',503);event(gas,'pointerup',504);check(a.input().steer===0&&a.input().throttle===0,'Release left touch held');
      rows.push({viewport:[win().innerWidth,win().innerHeight],layout,leftHanded,before,neutral,resumed,after,captures,clockStopped:true});
     }finally{for(const restore of restored)restore();a.clearInput();a.pause();}
    }
    return {rows,scope:'Complete injected touch contact + ordinary native button activation journeys, portrait/landscape, both layouts/sides. Pointer capture is explicitly stubbed for synthetic contacts; actual trusted pointer/canvas checks are separate, and physical touch is excluded.'};
   }finally{Object.assign(a.settings,settings);a.applyPreferences();frame.style.width='1280px';frame.style.height='720px';}
  },
  'Real IndexedDB termination before and after commit':async()=>{
   const prefix=frame.contentDocument.querySelector('base').href;
   const {openStore}=await import(prefix+'storage.js');
   const database='scrap-audit-termination-'+crypto.randomUUID();
   const track=structuredClone(api().content.tracks[0].track);track.name='Last committed draft';
   const observations=[];
   const terminateAt=phase=>new Promise((resolve,reject)=>{
    const worker=new Worker('/__audit__/storage-worker.js',{type:'module'});
    const events=[];let complete=false;
    const timer=setTimeout(()=>{worker.terminate();reject(Error('Termination fixture did not establish '+phase+' boundary'));},10000);
    worker.onerror=e=>{clearTimeout(timer);worker.terminate();reject(Error(e.message));};
    worker.onmessage=({data})=>{
     events.push(data);
     if(data.event==='fixture-error'||data.event.startsWith('unexpected')){
      clearTimeout(timer);worker.terminate();reject(Error('Unqualified termination boundary: '+JSON.stringify(events)));return;
     }
     if(data.event===(phase==='before'?'request-succeeded-transaction-incomplete':'transaction-committed')){
      complete=true;worker.terminate();clearTimeout(timer);resolve({phase,events,terminatedAfterQualifiedSignal:complete});
     }
    };
    worker.postMessage({prefix,database,track,phase});
   });
   observations.push(await terminateAt('before'));
   let store=await openStore({name:database}),record=await store.get('draft:'+track.id);
   check(record.track.name==='Last committed draft'&&record.revision===1,'Uncommitted transaction survived termination or previous draft was lost');store.close();
   observations.push(await terminateAt('after'));
   store=await openStore({name:database});record=await store.get('draft:'+track.id);
   check(record.track.name==='Committed before termination'&&record.revision===2,'Committed transaction lost after termination');store.close();
   return {database,observations,lastCommittedSourceSurvived:true,
    scope:'Actual browser IndexedDB and Store.saveDraft, terminated Worker execution contexts with qualified transaction events. This is not an OS/browser-process crash or storage-device power-loss guarantee.'};
  },
  'Recreate twelve official circuits in Workshop':async()=>{
   const originals=api().content.tracks.map(item=>structuredClone(item.track));
   const exports=[];
   for(const original of originals){
    trace.length=0;await blank(original.name);
    change('#track-scene',original.scenePreset);change('#track-laps',original.laps);
    for(let i=0;i<original.pieces.length;i++){
     const p=original.pieces[i];change('#piece-kind',p.kind);
     if(['straight','start','taper'].includes(p.kind))change('#piece-length',p.length);
     if(['quarter','hairpin'].includes(p.kind)){change('#piece-radius',p.radius);change('#piece-turn',p.turn);}
     activate('place-road');
     if(i===0)numericFields({x:p.x,z:p.z,heading:p.heading*180/Math.PI});
    }
    activate('close-loop');activate('setup-race');
    for(const patch of original.patches){
     const index=original.pieces.findIndex(p=>p.id===patch.pieceId);selectRoad(index);
     activate('partial-patch');
     for(const key of ['surface','start','end','lateralMin','lateralMax']){
      const el=doc().querySelector('[data-patch="'+key+'"]');check(el,'Missing patch field '+key);el.value=String(patch[key]);
      trace.push({action:'patch-field',key,value:patch[key]});
     }
     activate('patch-apply');
    }
    for(const prop of original.props){
     const prefix=frame.contentDocument.querySelector('base').href;
     const {compile,nearest}=await import(prefix+'track.js');
     const road=nearest(compile(original),prop.x,prop.z);
     const index=original.pieces.findIndex(p=>p.id===road?.pieceId);
     check(index>=0,'Object blueprint has no nearest road');selectRoad(index);change('#editor-tool',prop.kind);activate('add-prop');
     numericFields({x:prop.x,z:prop.z,heading:prop.heading*180/Math.PI,length:prop.length,width:prop.width});
    }
    const w=api().workshop;await w.flush();check(w.compiled.valid,original.slug+' recreated geometry invalid: '+JSON.stringify(w.compiled.problems));
    check(w.track.pieces.length===original.pieces.length,'Road count changed');
    check(w.track.gates.length===original.gates.length&&w.track.spawns.length===original.spawns.length,'Race setup count changed');
    for(let i=0;i<original.pieces.length;i++)for(const key of ['kind','x','z','heading','width','exitWidth',...(['quarter','hairpin'].includes(original.pieces[i].kind)?['radius','turn']:['length'])]){
     const a=w.track.pieces[i][key],b=original.pieces[i][key];check(typeof a==='number'?Math.abs(a-b)<1e-8:a===b,original.slug+' UI shape differs at '+i+'/'+key);
    }
    let exported=null;const originalDownload=w.download;
    try{w.download=(filename,html,type)=>{exported={filename,html,type};};activate('work-file');await waitFor(()=>doc().querySelector('dialog').open&&doc().querySelector('[data-file="export"]'));doc().querySelector('[data-file="export"]').click();}
    finally{w.download=originalDownload;}
    check(exported?.html?.includes('urn:ScrapLeague:Track'),'Ordinary export control did not produce semantic HTML');
    const record={utc:new Date().toISOString(),slug:original.slug,entry:frame.contentWindow.location.href,release:frame.contentDocument.querySelector('base').href.split('/').filter(Boolean).at(-1),
     method:'Browser-executed native form/button commands from New blank track; geometry never assigned directly. Synthetic form activation; not physical keyboard/touch proof.',
     originalCreation:'Historical CLI; this is a new Workshop recreation and export, not a claim about original history.',
     pendingBinding:'Stable resource IDs/catalog metadata and inactive piece parameters require independently reviewed binding before reuse as official sources.',
     trace:[...trace],exported};
    const response=await fetch('/__audit__/exports',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(record)});check(response.ok,'Export receipt not retained');
    exports.push({slug:original.slug,pieces:w.track.pieces.length,patches:w.track.patches.length,props:w.track.props.length,gates:w.track.gates.length,actions:trace.length});
   }
   return {exports,scope:'Twelve present-tense UI recreations/exports; stable binding and independent semantic comparison recorded separately. This does not repair historical CLI provenance.'};
  },
 };
}
